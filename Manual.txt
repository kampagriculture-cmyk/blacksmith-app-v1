5.โครงเต็มคือ 4 ตารางแยกกัน เชื่อมด้วย production_log_id:
production_logs      id, machine_id, knife_id, lot_no, status,
                      ended_at, total_qty, good_qty,
                      operator_id, supervisor_id, qc_approved, remark,
                      created_at, updated_at

stone_changes         id, production_log_id (FK → production_logs.id),
                      qty_before_change, size_left, size_right,
                      downtime_start, downtime_end
                      -- 1 log มี stone_change ได้ 0 หรือ 1 แถว

tune_rounds            id, production_log_id (FK → production_logs.id),
                      round_no, start_time, end_time
                      -- 1 log มี tune_rounds ได้ 0 หรือหลายแถว (หลายรอบจูน)

defect_entries         id, production_log_id (FK → production_logs.id),
                      defect_type_code (FK → defect_types.code), qty
                      -- 1 log มี defect_entries ได้ 0 ถึง 15 แถว
